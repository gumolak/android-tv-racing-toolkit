cmake_minimum_required(VERSION 3.16)
project(android_tv_racing_toolkit LANGUAGES C CXX)
include(CTest)
add_library(racing_policies INTERFACE)
target_include_directories(racing_policies INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(racing_policies INTERFACE cxx_std_11)
option(RACING_TOOLKIT_WITH_SDL2 "Build the real SDL2 registry and raw-input example" OFF)
if(RACING_TOOLKIT_WITH_SDL2)
  find_package(SDL2 CONFIG REQUIRED)
  add_library(controller_order STATIC src/ControllerIdentity.c)
  target_include_directories(controller_order PUBLIC include)
  target_link_libraries(controller_order PUBLIC SDL2::SDL2)
  add_executable(bind_two_controllers examples/bind_two_controllers.c)
  target_link_libraries(bind_two_controllers PRIVATE controller_order)
endif()
if(BUILD_TESTING)
  add_executable(policy_tests tests/test_policies.cpp)
  target_link_libraries(policy_tests PRIVATE racing_policies)
  add_test(NAME policy_tests COMMAND policy_tests)
  add_executable(controller_order_tests src/ControllerIdentity.c tests/test_controller_order.c)
  target_include_directories(controller_order_tests PRIVATE tests/sdl_stub include)
  add_test(NAME controller_order_tests COMMAND controller_order_tests)
endif()
